# BEGIN RosmarOps STATIC
# Versioned in qtorb/rosmarops-web (wp/htaccess-static.txt). Prepended once by the deploy (GTM-114).
# «About RosmarOps» in EN, ES and CA is served from the repository (site/_static/), not from WordPress, at the same
# addresses. [END] stops rewriting there, so WordPress never sees these requests. Plain http is left to the HTTPS
# block (the conditions below skip it), and site/_static/ itself is not reachable from outside.
<IfModule mod_rewrite.c>
RewriteEngine On
RewriteRule ^_static/ - [R=404,L]
RewriteCond %{SERVER_PORT} !^80$ [OR]
RewriteCond %{HTTP:X-Forwarded-Proto} https
RewriteRule ^blog/about/$ _static/about-en.html [END]
RewriteCond %{SERVER_PORT} !^80$ [OR]
RewriteCond %{HTTP:X-Forwarded-Proto} https
RewriteRule ^es/blog/que-es-rosmarops/$ _static/about-es.html [END]
RewriteCond %{SERVER_PORT} !^80$ [OR]
RewriteCond %{HTTP:X-Forwarded-Proto} https
RewriteRule ^ca/blog/que-es-rosmarops/$ _static/about-ca.html [END]
</IfModule>
# END RosmarOps STATIC
